cmake_minimum_required(VERSION 3.20)
project(NaughtyBearRecomp CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if (NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "The recompiled code requires Clang (configure with CC=clang CXX=clang++).")
endif()

# XEX loader (XenonUtils) for the host runtime.
add_subdirectory(tools/XenonRecomp EXCLUDE_FROM_ALL)

set(GENERATED_DIR ${CMAKE_SOURCE_DIR}/generated/ppc)
if (NOT EXISTS ${GENERATED_DIR}/ppc_func_mapping.cpp)
    message(FATAL_ERROR "No recompiled code in generated/ppc. Run tools/recompile.sh first.")
endif()

# The recompiled game code. Regenerated by tools/recompile.sh, so re-run CMake after recompiling.
file(GLOB RECOMP_SOURCES CONFIGURE_DEPENDS ${GENERATED_DIR}/ppc_recomp.*.cpp)
add_library(NaughtyBearRecompLib STATIC ${RECOMP_SOURCES} ${GENERATED_DIR}/ppc_func_mapping.cpp)
target_include_directories(NaughtyBearRecompLib PUBLIC
    ${GENERATED_DIR}
    ${CMAKE_SOURCE_DIR}/tools/XenonRecomp/thirdparty/simde)
target_compile_options(NaughtyBearRecompLib PUBLIC -march=x86-64-v3)
target_compile_options(NaughtyBearRecompLib PRIVATE
    -fno-strict-aliasing
    -Wno-unused-label -Wno-unused-variable -Wno-unused-but-set-variable)

# Logging stubs for every import the runtime does not implement yet.
file(GLOB_RECURSE RUNTIME_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.cpp)
list(FILTER RUNTIME_SOURCES EXCLUDE REGEX "src/gpu/fallback/|src/gpu/translator/")

set(IMPORT_STUBS ${CMAKE_BINARY_DIR}/import_stubs.cpp)
add_custom_command(
    OUTPUT ${IMPORT_STUBS}
    COMMAND ${CMAKE_SOURCE_DIR}/tools/gen_import_stubs.py ${GENERATED_DIR}/ppc_func_mapping.cpp ${IMPORT_STUBS}
    DEPENDS ${CMAKE_SOURCE_DIR}/tools/gen_import_stubs.py ${GENERATED_DIR}/ppc_func_mapping.cpp ${RUNTIME_SOURCES}
    COMMENT "Generating import stubs")

find_package(SDL2 REQUIRED)

# plume: the renderer abstraction Unleashed Recompiled uses, here on Vulkan through SDL.
set(SDL_VULKAN_ENABLED ON CACHE BOOL "" FORCE)
add_subdirectory(thirdparty/plume EXCLUDE_FROM_ALL)

# zstd and smol-v decode the shader cache; both come with XenosRecomp.
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(tools/XenosRecomp/thirdparty/zstd/build/cmake EXCLUDE_FROM_ALL)
add_library(smolv STATIC tools/XenosRecomp/thirdparty/smol-v/source/smolv.cpp)
target_include_directories(smolv PUBLIC tools/XenosRecomp/thirdparty/smol-v/source)

# The shader translator, linked into the game: XenosRecomp's sources (patched,
# tools/apply_patches.sh) and DXC, which ships with XenosRecomp and is copied next to the
# executable. Shaders are translated when the game first runs (src/gpu/shader_store.cpp).
set(XENOS_DIR ${CMAKE_SOURCE_DIR}/tools/XenosRecomp/XenosRecomp)
set(DXC_DIR ${CMAKE_SOURCE_DIR}/tools/XenosRecomp/thirdparty/dxc-bin)
set(SHADER_COMMON_TEXT ${CMAKE_BINARY_DIR}/translator/shader_common_text.h)
file(READ ${XENOS_DIR}/shader_common.h SHADER_COMMON_CONTENTS)
file(GLOB TRANSLATOR_PATCHES ${CMAKE_SOURCE_DIR}/patches/XenosRecomp/*.patch)
set(TRANSLATOR_PATCH_CONTENTS "")
foreach(patch ${TRANSLATOR_PATCHES})
    file(READ ${patch} contents)
    string(APPEND TRANSLATOR_PATCH_CONTENTS "${contents}")
endforeach()
string(SHA1 TRANSLATOR_PATCH_HASH "${TRANSLATOR_PATCH_CONTENTS}")
file(WRITE ${SHADER_COMMON_TEXT}.tmp "#pragma once\nstatic const char SHADER_COMMON_TEXT[] = R\"NBHLSL(${SHADER_COMMON_CONTENTS})NBHLSL\";\nstatic const char TRANSLATOR_PATCH_HASH[] = \"${TRANSLATOR_PATCH_HASH}\";\n")
configure_file(${SHADER_COMMON_TEXT}.tmp ${SHADER_COMMON_TEXT} COPYONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${XENOS_DIR}/shader_common.h ${TRANSLATOR_PATCHES})

add_library(dxcompiler SHARED IMPORTED)
set_target_properties(dxcompiler PROPERTIES IMPORTED_LOCATION ${DXC_DIR}/lib/x64/libdxcompiler.so)
target_include_directories(dxcompiler INTERFACE ${DXC_DIR}/inc)

add_library(nb_shader_translator STATIC
    src/gpu/translator/shader_translator.cpp
    ${XENOS_DIR}/shader_recompiler.cpp
    ${XENOS_DIR}/dxc_compiler.cpp)
target_include_directories(nb_shader_translator PRIVATE ${XENOS_DIR} ${CMAKE_BINARY_DIR}/translator)
target_precompile_headers(nb_shader_translator PRIVATE ${XENOS_DIR}/pch.h)
target_compile_options(nb_shader_translator PRIVATE -Wno-switch -Wno-unused-variable -Wno-null-arithmetic -fms-extensions)
target_link_libraries(nb_shader_translator PUBLIC dxcompiler PRIVATE fmt::fmt xxHash::xxhash smolv libzstd_static)

add_library(o1heap STATIC thirdparty/o1heap/o1heap.c)
target_include_directories(o1heap PUBLIC thirdparty/o1heap)

# NB_TRACE_D3D: log the game's calls into its Direct3D library (docs/d3d_map.md).
option(NB_TRACE_D3D "Trace calls into the game's Direct3D library" OFF)
set(TRACE_WRAPPERS)
if (NB_TRACE_D3D)
    set(TRACE_WRAPPERS ${CMAKE_BINARY_DIR}/d3d_trace_wrappers.cpp)
    add_custom_command(
        OUTPUT ${TRACE_WRAPPERS}
        COMMAND ${CMAKE_SOURCE_DIR}/tools/gen_trace_wrappers.py ${GENERATED_DIR}/ppc_func_mapping.cpp ${TRACE_WRAPPERS} 82F1C000 82F50000
        DEPENDS ${CMAKE_SOURCE_DIR}/tools/gen_trace_wrappers.py ${GENERATED_DIR}/ppc_func_mapping.cpp
        COMMENT "Generating Direct3D trace wrappers")
endif()

# The renderer's own shaders (copies for resolves and presentation), compiled to SPIR-V
# headers with glslangValidator.
find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
set(HOST_SHADER_DIR ${CMAKE_BINARY_DIR}/host_shaders)
set(HOST_SHADER_HEADERS)
foreach(shader blit.vert:g_blitVS blit.frag:g_blitPS)
    string(REPLACE ":" ";" parts ${shader})
    list(GET parts 0 file)
    list(GET parts 1 name)
    set(header ${HOST_SHADER_DIR}/${file}.h)
    add_custom_command(
        OUTPUT ${header}
        COMMAND ${CMAKE_COMMAND} -E make_directory ${HOST_SHADER_DIR}
        COMMAND ${GLSLANG_VALIDATOR} -V --target-env vulkan1.1 --vn ${name} -o ${header} ${CMAKE_SOURCE_DIR}/src/gpu/shaders/${file}
        DEPENDS ${CMAKE_SOURCE_DIR}/src/gpu/shaders/${file}
        COMMENT "Compiling ${file}")
    list(APPEND HOST_SHADER_HEADERS ${header})
endforeach()

# The runtime: ported from Unleashed Recompiled (GPL-3.0), see NOTICE.md.
add_executable(NaughtyBear ${RUNTIME_SOURCES} ${IMPORT_STUBS} ${TRACE_WRAPPERS} ${HOST_SHADER_HEADERS})
if (NB_TRACE_D3D)
    target_compile_definitions(NaughtyBear PRIVATE NB_TRACE_D3D)
endif()
target_include_directories(NaughtyBear PRIVATE
    src
    src/gpu
    thirdparty/unordered_dense/include
    tools/XenonRecomp/thirdparty/xxHash
    tools/XenosRecomp/thirdparty/zstd/lib
    ${HOST_SHADER_DIR})
target_precompile_headers(NaughtyBear PRIVATE src/stdafx.h)

# FFmpeg decodes XMA audio (src/apu/xma.cpp): a minimal static build of the submodule,
# with only the XMA decoders.
include(ExternalProject)
set(FFMPEG_PREFIX ${CMAKE_BINARY_DIR}/ffmpeg)
ExternalProject_Add(ffmpeg_build
    SOURCE_DIR ${CMAKE_SOURCE_DIR}/thirdparty/FFmpeg
    BINARY_DIR ${FFMPEG_PREFIX}/build
    INSTALL_DIR ${FFMPEG_PREFIX}
    PATCH_COMMAND sh -c "git apply --reverse --check ${CMAKE_SOURCE_DIR}/patches/FFmpeg/*.patch 2>/dev/null || git apply ${CMAKE_SOURCE_DIR}/patches/FFmpeg/*.patch"
    CONFIGURE_COMMAND ${CMAKE_SOURCE_DIR}/thirdparty/FFmpeg/configure --prefix=<INSTALL_DIR> --cc=${CMAKE_C_COMPILER}
        --disable-everything --disable-programs --disable-doc --disable-network --disable-autodetect --disable-asm
        --disable-debug --enable-decoder=xma1,xma2 --disable-avdevice --disable-avformat --disable-swscale
        --disable-swresample --disable-postproc --disable-avfilter --enable-static --disable-shared --enable-pic
    BUILD_COMMAND make -j
    INSTALL_COMMAND make install
    BUILD_BYPRODUCTS ${FFMPEG_PREFIX}/lib/libavcodec.a ${FFMPEG_PREFIX}/lib/libavutil.a)
file(MAKE_DIRECTORY ${FFMPEG_PREFIX}/include)
add_library(ffmpeg INTERFACE)
add_dependencies(ffmpeg ffmpeg_build)
target_include_directories(ffmpeg INTERFACE ${FFMPEG_PREFIX}/include)
target_link_libraries(ffmpeg INTERFACE ${FFMPEG_PREFIX}/lib/libavcodec.a ${FFMPEG_PREFIX}/lib/libavutil.a m)
target_compile_definitions(NaughtyBear PRIVATE NB_HAVE_FFMPEG)
target_link_libraries(NaughtyBear PRIVATE ffmpeg)
target_link_libraries(NaughtyBear PRIVATE nb_shader_translator)
set_target_properties(NaughtyBear PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
add_custom_command(TARGET NaughtyBear POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${DXC_DIR}/lib/x64/libdxcompiler.so $<TARGET_FILE_DIR:NaughtyBear>)
target_link_libraries(NaughtyBear PRIVATE NaughtyBearRecompLib XenonUtils o1heap fmt::fmt xxHash::xxhash plume SDL2::SDL2 libzstd_static smolv)
